\begin{tikzpicture}[
  font=\small, node distance=5mm and 8mm,
  box/.style={draw, rounded corners=2pt, align=left, text width=84mm, inner sep=5pt, fill=white},
  wide/.style={box, text width=98mm},
  model/.style={box, fill=blue!6},

  post/.style={box, fill=green!8},
  dec/.style={draw, diamond, aspect=2.6, align=center, inner sep=1.5pt, fill=white},
  note/.style={draw=gray, dashed, rounded corners=2pt, align=left, text width=58mm, font=\footnotesize, inner sep=4pt, fill=gray!5},
  arr/.style={-{Stealth[length=2mm]}, thick},
  lab/.style={font=\footnotesize, fill=white, inner sep=1.5pt},
]
\node[wide] (start) {\textbf{Start.} vv.\,1--12 intact and fixed; vv.\,13--29 from a starting text. Sixteen chains start from Eric Cullhed's eight reconstructions (two per start, variants of one text: 9\,\% of restored words differ); twenty-four more from \emph{unrelated} starts, late states of earlier chains picked greedily so that every pair differs in 89--100\,\% of its restored words (\texttt{spread\_starts.py}). 10\,000 steps per chain, 100 warm steps, burn-in 300, a state recorded every 10 steps; run as four 2\,500-step segments, each continuing the last state of the one before.};
\node[box, below=of start] (pick) {\textbf{Propose a site.} Draw a damaged verse $v \in \{13,\dots,29\}$ uniformly and a letter-block span in its restored part (a mixture of block widths, or the whole restored part).};
\node[model, below=of pick] (sto) {\textbf{Stoicheia proposes.} The span is masked; 16 replacements are drawn by iterative unmasking (8 steps, sampling temperature 0.65) conditioned on the whole passage, the papyrus prefix of $v$ kept.};
\node[box, below=of sto] (filt) {\textbf{Formal filters} (hard, in this order). Papyrus prefix kept \;$\cdot$\; 26--34 letters \;$\cdot$\; every word real Greek (lexicon + letter model) \;$\cdot$\; \textbf{elision rule}: a remnant ($\delta$' $\tau$' $\gamma$' $\theta$' $\tau\alpha\tilde\upsilon\tau$' \dots) only before a vowel, tenuis never before a rough breathing, derived aspirate only before one \;$\cdot$\; particle-pair blacklist \;$\cdot$\; tragic register above the Sophocles 5th percentile \;$\cdot$\; no content word or lemma repeated within $\pm2$ verses \;$\cdot$\; scans as an iambic trimeter without hiatus \;$\cdot$\; distinct from the current line.};
\node[dec, below=of filt] (any) {any survivor?};
\node[box, below=of any] (draw) {\textbf{One candidate} is drawn uniformly from the survivors (Metropolis-within-Gibbs: the target is Stoicheia's own distribution under the constraints).};
\node[box, below=of draw] (formal) {\textbf{Formal term.} \\[2pt] $\Delta_{\text{formal}} = \dfrac{1}{T_{\text{formal}}}\,\Delta \log\text{-odds}(\text{Sophocles}) + \Delta_{\text{finite}} + \Delta_{\text{particles}} + \Delta_{\text{repetition}}$ \\[3pt] with $1/T_{\text{formal}} = 0$: style-blind, so the walk remains a valid imputation source. The three calibration terms move the restorer's own rates to Sophocles': $\operatorname{logit} p - \operatorname{logit} q$ per verse with a finite verb ($+0.02$), per particle ($-0.53$), and $\log(p_b/q_b)$ per word by the band $b$ of its nearest earlier occurrence (same line $+0.02$, 1 back $+0.12$, 2 back $+0.24$, 3--4 back $-0.40$, 5--7 back $-0.10$; weight $\tfrac12$).};
\node[judge, below=of formal] (jud) {\textbf{Coherence judge} (Qwen3-235B-A22B-Instruct-2507, FP8, 22B active parameters per token, split over the four GH200 GPUs of one node). Window of $\pm2$ verses on the elided surface (elision marks restored, no punctuation), old versus new line as versions A and B: \emph{``in which version is line $v$ the more coherent continuation, in grammar, syntax and meaning?''} Asked in both orders: \\[2pt] $\Delta_{\text{judge}} = \tfrac12\,(\ell_{\text{new first}} - \ell_{\text{old first}})$, \; $\ell$ the next-token log-odds of answering A, \\[2pt] i.e.\ the order-averaged log-odds for the new line.};
\node[judge, below=of jud] (acc) {\textbf{Accept} with probability \\[2pt] $\min\bigl\{1,\ \exp(\Delta_{\text{formal}} + \Delta_{\text{judge}}/T_{\text{judge}})\bigr\}$ \\[2pt] with $1/T_{\text{judge}} = 4$ for every set in the report.};
\node[dec, below=of acc] (ok) {accepted?};
\node[box, below=of ok, text width=60mm] (upd) {\textbf{Update} verse $v$, its lemmas, finite flag, particle count, repetition bands and register; log the step with $\Delta_{\text{judge}}$.};
\draw[arr] (start) -- (pick);

\draw[arr] (formal) -- (jud);
\draw[arr] (jud) -- (acc);
\draw[arr] (acc) -- (ok);
\draw[arr] (ok) -- node[lab, right] {yes} (upd);

% side notes, and the pipeline after the walk under them
\node[note, right=12mm of rail, anchor=west] at (sto.east -| rail) (n1) {Iota adscript on the letter level (the papyrus and Stoicheia); converted to subscript for the style space, the judge and the lemmata.};
\node[note, below=of n1] (n2) {Typically a few of the 16 proposals survive the filters. The 235B judge accepted 11\,\% of the judged proposals (10\,\% of all steps); the 72B 9\,\%; Claude Sonnet 5.5 at low effort 46\,\%, nearly indifferent.};
\node[note, below=of n2] (n3) {Plain walk: $1/T_{\text{judge}} = 0$ (judge consulted and logged only). Tilted chains ($1/T_{\text{formal}} = 1$) select for Sophoclean style and are not an imputation source.};
\node[note, below=of n3] (n4) {Rule set v3e for every set here (v3c + elision rule, v3d + particle term, v3e + repetition term). Earlier rule sets let the walk exploit a lax elided surface and are dropped from the dataset.};
\node[note, below=of n4] (n5) {Optional economies for a priced judge (the Claude pilot only, off for the Qwen sets): proposals changing under 10\,\% of a verse's letters are dropped unjudged; burn-in ends when 50\,\% of the restored letters differ from the start; a state is recorded when 10\,\% have changed since the last one.};
\node[note, below=of n5] (n6) {Mixing: after 10\,000 steps the chains' final states differ in 96\,\% of their restored words whether or not they shared a start, yet 22--28 of 34 topics have $\hat R > 1.1$; the pooled lift averages over chains that have not met.};
\node[post, below=8mm of n6, text width=58mm, font=\footnotesize] (parse) {\textbf{After the walk: from states to documents.} Every recorded state is lemmatised by the morphological parser; the lemmata are filtered to the 3\,128-lemma content vocabulary and reduced to a \emph{bag of topic-bearing lemmata} (34 word2vec clusters, iota subscript normalised). One state is one document, about 30 topic lemmata; the sets go to the dataset \texttt{Urdatorn/bsda-miniproject-2026} with their chain set, chain and step.};
\node[post, below=of parse, text width=58mm, font=\footnotesize] (model) {\textbf{Bayesian models.} \emph{Hard}: Dirichlet--multinomial per document with Sophocles' profile as a one-lemma prior, exact posterior; the imputations' mean profile against windows of seventeen Sophoclean trimeters (and the other tragedians, Classical Attic), lift with a 90\,\% interval from resampling chains and windows. \emph{Soft}: LDA with the topic--word matrix fixed, per-document NUTS. \emph{Checks}: drift of the lift in 1\,000-step windows, Gelman--Rubin $\hat R$ across chains, the sets from Eric's starts against the sets from unrelated starts, and the two Qwen judges against a Claude Sonnet 5.5 pilot.};
\draw[arr] (upd.south) -- ++(0,-6mm) -| node[lab, pos=0.25, below] {after the last step} (parse.west);
\draw[arr] (parse) -- (model);
